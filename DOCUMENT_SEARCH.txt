Task c: Group & Find Users Sharing the Same Firstname
e_shopping> db.student.aggregate([
  {
    $group: {
      _id: "$firstname",
      count: { $sum: 1 },
      users: { $push: "$$ROOT" }
    }
  },
  {
    $match: {
      count: { $gt: 1 }
    }
  }
])

// Result Output:
[
  {
    _id: 'adhi',
    count: 2,
    users: [
      { _id: 1, firstname: 'adhi', lastname: 'ravi', age: 16, email: 'adhiravi@gmail.com' },
      { _id: 2, firstname: 'adhi', lastname: 'mani', age: 12, email: 'dhivya@gmail.com' }
    ]
  }
]
