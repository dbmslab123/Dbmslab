Step 2: Aggregate Function - $sum
e_shopping> db.s_mark.aggregate([
  {
    $addFields: {
      totalmark: { $sum: ["$java", "$cn", "$os"] }
    }
  }
])

// Result of $sum:
[
  { _id: 1, sname: 'Arun', java: 76, cn: 73, os: 98, totalmark: 247 },
  { _id: 2, sname: 'banu', java: 67, cn: 56, os: 76, totalmark: 199 }
]

Step 3: Multi-Operator Aggregation ($add, $multiply, $divide)
e_shopping> db.s_mark.aggregate([
  {
    $addFields: {
      totalmark: { $add: ["$java", "$cn", "$os"] },
      percentage: {
        $multiply: [
          { $divide: [{ $add: ["$java", "$cn", "$os"] }, 300] },
          100
        ]
      }
    }
  }
])

// Result Output:
[
  {
    _id: 1,
    sname: 'Arun',
    java: 76,
    cn: 73,
    os: 98,
    totalmark: 247,
    percentage: 82.33333333333334
  },
  {
    _id: 2,
    sname: 'banu',
    java: 67,
    cn: 56,
    os: 76,
    totalmark: 199,
    percentage: 66.33333333333333
  }
]

Step 9: Aggregate Stage - $group
e_shopping> db.s_mark.aggregate([
  {
    $group: {
      _id: "$edu_status",
      avg_java: { $avg: "$java" },
      avg_cn: { $avg: "$cn" },
      avg_os: { $avg: "$os" },
      count: { $sum: 1 }
    }
  }
])


Result of $group Stage
[
  { _id: 'intern', avg_java: 70, avg_cn: 50.5, avg_os: 72.5, count: 2 },
  { _id: 'regular', avg_java: 82.5, avg_cn: 70, avg_os: 76, count: 2 }
]
